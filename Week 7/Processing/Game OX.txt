board = [['' for _ in range(3)] for _ in range(3)]
current_player = 'X'
winner = None
game_over = False

def setup():
    size(300, 350)
    textAlign(CENTER, CENTER)

def draw():
    background(240)
    draw_grid()
    draw_board()
    draw_status()

def draw_grid():
    stroke(0)
    strokeWeight(4)
    line(100, 0, 100, 300)
    line(200, 0, 200, 300)
    line(0, 100, 300, 100)
    line(0, 200, 300, 200)

def draw_board():
    textSize(48)
    for r in range(3):
        for c in range(3):
            if board[r][c] == 'X':
                fill(200, 50, 50)
                text("X", c * 100 + 50, r * 100 + 45)
            elif board[r][c] == 'O':
                fill(50, 50, 200)
                text("O", c * 100 + 50, r * 100 + 45)

def draw_status():
    fill(0)
    textSize(16)
    if winner:
        text("Player " + winner + " Wins! Click to restart", width / 2, 325)
    elif game_over:
        text("Draw Game! Click to restart", width / 2, 325)
    else:
        text("Turn: Player " + current_player, width / 2, 325)

def mousePressed():
    global current_player, winner, game_over
    if game_over or winner:
        reset_game()
        return

    if mouseY < 300:
        col = mouseX // 100
        row = mouseY // 100
        if board[row][col] == '':
            board[row][col] = current_player
            check_win()
            if not winner and not check_tie():
                current_player = 'O' if current_player == 'X' else 'X'

def check_win():
    global winner, game_over
    for i in range(3):
        if board[i][0] == board[i][1] == board[i][2] != '':
            winner = board[i][0]
            game_over = True
        if board[0][i] == board[1][i] == board[2][i] != '':
            winner = board[0][i]
            game_over = True
    if board[0][0] == board[1][1] == board[2][2] != '':
        winner = board[0][0]
        game_over = True
    if board[0][2] == board[1][1] == board[2][0] != '':
        winner = board[0][2]
        game_over = True

def check_tie():
    global game_over
    for r in range(3):
        for c in range(3):
            if board[r][c] == '':
                return False
    game_over = True
    return True

def reset_game():
    global board, current_player, winner, game_over
    board = [['' for _ in range(3)] for _ in range(3)]
    current_player = 'X'
    winner = None
    game_over = False
